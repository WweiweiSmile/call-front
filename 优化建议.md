#### 方案 B - 使用 useContext + useReducer

```typescript
// src/store/AppContext.tsx
import { createContext, useContext, useReducer } from 'react';

const AppContext = createContext<AppContextType | null>(null);

type AppAction = 
  | { type: 'SET_GAMES'; payload: Game[] }
  | { type: 'SET_USER'; payload: User | null };

function appReducer(state: AppState, action: AppAction): AppState {
  switch (action.type) {
    case 'SET_GAMES':
      return { ...state, games: action.payload };
    default:
      return state;
  }
}

export function AppProvider({ children }: { children: React.ReactNode }) {
  const [state, dispatch] = useReducer(appReducer, initialState);
  return (
    <AppContext.Provider value={{ state, dispatch }}>
      {children}
    </AppContext.Provider>
  );
}

export function useAppContext() {
  const context = useContext(AppContext);
  if (!context) {
    throw new Error('useAppContext must be used within AppProvider');
  }
  return context;
}
```

```typescript
import { useCallback } from 'react';
import { showToast } from '@tarojs/taro';

export function useErrorHandler() {
  const handleError = useCallback((error: unknown) => {
    console.error('Error:', error);
    
    if (error instanceof ApiError) {
      showToast({ title: error.message, icon: 'none' });
    } else if (error instanceof Error) {
      showToast({ title: error.message, icon: 'none' });
    } else {
      showToast({ title: '发生未知错误', icon: 'none' });
    }
  }, []);

  return { handleError };
}
```

### 5. 游戏详情页轮询优化

**位置：** `src/pages/game-detail/index.tsx`

**问题：**
- 5秒轮询可能太频繁
- 缺少请求取消机制
- 游戏结束后应该延长轮询间隔

**优化方案：**

```typescript
const POLL_INTERVAL_ACTIVE = 10000; // 10秒
const POLL_INTERVAL_ENDED = 30000;  // 30秒

const loadData = useCallback(async (showLoading = false) => {
  const controller = new AbortController();
  loadDataControllerRef.current = controller;

  try {
    // 请求逻辑...
  } catch (error) {
    if (error.name === 'AbortError') {
      return;
    }
    // 错误处理...
  }
}, [gameId]);

useEffect(() => {
  if (!gameId) return;

  let pollingTimer: NodeJS.Timeout;

  const poll = async () => {
    if (!isVisible || !isGameEnded) {
      await loadData(false);
    }

    const interval = isGameEnded ? POLL_INTERVAL_ENDED : POLL_INTERVAL_ACTIVE;
    pollingTimer = setTimeout(poll, interval);
  };

  poll();

  return () => {
    if (pollingTimer) {
      clearTimeout(pollingTimer);
    }
    if (loadDataControllerRef.current) {
      loadDataControllerRef.current.abort();
    }
  };
}, [gameId, isVisible, isGameEnded, loadData]);
```

### 6. 代码重复 - 提取共用游戏列表 Hook

**位置：** `src/pages/games/index.tsx`, `src/pages/my-games/index.tsx`

**优化方案：**

```typescript
// src/hooks/useGameList.ts
import { useMemo } from 'react';
import { useLoadMore } from './useLoadMore';

type GameListType = 'all' | 'my';

export function useGameList(type: GameListType) {
  const { data, loading, refresh, loadMore, hasMore } = useLoadMore(
    async (page) => {
      if (type === 'all') {
        return await gameApi.getGames({ page, pageSize: 20 });
      } else {
        return await gameApi.getMyGames({ page, pageSize: 20 });
      }
    },
    [type]
  );

  const filteredGames = useMemo(() => {
    const ongoing = data.filter(g => g.status === 'ongoing');
    const pending = data.filter(g => g.status === 'pending');
    const ended = data.filter(g => g.status === 'ended');
    return { ongoing, pending, ended, all: data };
  }, [data]);

  return {
    games: filteredGames,
    loading,
    refresh,
    loadMore,
    hasMore,
  };
}
```

```typescript
  items: T[];
  height: number;
  itemHeight: number;
  renderItem: (item: T, index: number) => React.ReactNode;

export function VirtualList<T>({ items, height, itemHeight, renderItem }: VirtualListProps<T>) {

      height={height}
      itemCount={items.length}
```

### 9. 安全性增强

#### 9.1 安全的 Token 存储

```typescript
// src/utils/secureStorage.ts
const SECURE_KEY = 'call_app_';

export const secureStorage = {
  setToken(token: string) {
    try {
      localStorage.setItem(`${SECURE_KEY}token`, token);
    } catch (e) {
      console.error('Failed to store token');
    }
  },

  getToken(): string | null {
    try {
      return localStorage.getItem(`${SECURE_KEY}token`);
    } catch {
      return null;
    }
  },

  removeToken() {
    try {
      localStorage.removeItem(`${SECURE_KEY}token`);
    } catch (e) {
      console.error('Failed to remove token');
    }
  },
};
```
